create or replace function atividade_log_update()
returns trigger
language plpgsql
as $$
	begin
	insert into tb_atividade_log
	(tx_query, nr_idade, tx_mensagem, cd_atividade)
	values(
		current_query(),
		extract (year from age(
				old.dt_nascimento	
				)),
		'MENSAGEM VIA TRIGGER',
		new.id_atividade
	);
	return new;
	end;
$$;

create trigger nome 
	before
	update on tb_atividade
for each row 
execute function atividade_log_update();

create sequence seq_pagamento_log start 1 increment 1;

create  TABLE tb_pagamentos_log (
    id_pagamento_log NUMERIC(18,0)
        DEFAULT nextval('seq_pagamento_log')
        PRIMARY KEY,
    tx_query VARCHAR(800),
    tx_forma VARCHAR(100),
    vl_antes NUMERIC,
    vl_depois NUMERIC,
    data_modificacao TIMESTAMP
);

CREATE OR REPLACE FUNCTION registrar_pagamento()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    NEW.tx_forma := LOWER(NEW.tx_forma);

    INSERT INTO tb_pagamentos_log
    VALUES (
        DEFAULT,
        CURRENT_QUERY(),
        NEW.tx_forma,
        old.vl_pagamento,
        NEW.vl_pagamento,
        CURRENT_TIMESTAMP
    );

    RETURN NEW;
END;
$$;

create trigger trigger_pagamentos
before insert on tb_pagamentos
for each row
execute function
registrar_pagamento();

create view vw_produto_mais_pedidos as
select 
 tb_produtos.id_produto,
 tb_produtos.tx_descricao,
 sum(tb_pedidos.vl_pago) as 
 total_valor_pago,
 max (tb_pedidos.data_pedidos) as 
 maior_data_pedido,
 max(tb_usuarios.id_usuario) as 
 maior_codigo_usuario
 from tb_pedidos
 inner join tb_usuarios
  on tb_pedidos.id_usuario =
  tb_usuarios.id_usuario
  inner join tb_produtos
  on tb_pedidos.id_produto =
  tb_pedidos.id_produto
  where tb_pedidos.status = 'pago'
  and tb_pedidos.fl_pago is true
  and tb_usuarios.tx_uf = 'sc'
  and tb_pedidos.vl_pago > 0
  group by 
  tb_produtos.id_produto,
  tb_produtos.tx_descricao
  order by total_valor_pago desc 
  limit 50;

create index idx_pedidos_status_pago on tb_pedidos (status.fl_pago);
