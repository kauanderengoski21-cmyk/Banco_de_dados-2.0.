select tb_categoria.tx_nome,
count(tb_produto.id_produto)
from tb_categoria 
inner join tb_produto
on tb_categoria.id_categoria  =
tb_produto.cd_categoria 
where tb_produto.fl_ativo = true
group by tb_categoria.tx_nome;

select tb_categoria.tx_nome
from tb_categoria
inner join tb_produto 
on tb_categoria.id_categoria =
tb_produto.cd_categoria 
group by tb_categoria.tx_nome 
having count (tb_produto.id_produto) > 2;
