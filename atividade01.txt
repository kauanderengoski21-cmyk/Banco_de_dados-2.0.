
SELECT 
    tb_cliente.tx_nome,
    SUM(
        tb_item_venda.nr_quantidade * tb_item_venda.vl_unitario
    ) AS total_gasto
FROM tb_cliente
INNER JOIN tb_venda
    ON tb_cliente.id_cliente = tb_venda.cd_cliente
INNER JOIN tb_item_venda
    ON tb_venda.id_venda = tb_item_venda.cd_venda
WHERE tb_venda.tx_status = 'pago'
GROUP BY tb_cliente.tx_nome
HAVING 
    SUM(
        tb_item_venda.nr_quantidade * tb_item_venda.vl_unitario
    ) > (
        SELECT AVG(total_gasto)
        FROM (
            SELECT 
                SUM(
                    tb_item_venda.nr_quantidade * tb_item_venda.vl_unitario
                ) AS total_gasto
            FROM tb_venda
            INNER JOIN tb_item_venda
                ON tb_venda.id_venda = tb_item_venda.cd_venda
            WHERE tb_venda.tx_status = 'pago'
            GROUP BY tb_venda.cd_cliente
        ) AS totais
    );


with totalComprasPorCliente as (
	SELECT 
	    tb_venda.cd_cliente,
	    SUM(
	        tb_item_venda.nr_quantidade * tb_item_venda.vl_unitario
	    ) AS total_gasto
	from  tb_venda
	INNER JOIN tb_item_venda
	    ON tb_venda.id_venda = tb_item_venda.cd_venda
	WHERE tb_venda.tx_status = 'PAGO'
	GROUP BY cd_cliente
),
media as (
select sum(total_gasto) / count(cd_cliente) as valor from totalComprasPorCliente
)
select * from tb_cliente
inner join totalComprasPorCliente on totalComprasPorCliente.cd_cliente  = tb_cliente.id_cliente 
inner join media on total_gasto > media.valor 

where 
	total_gasto > ( select media.valor from media)

select
tb_cliente.tx_nome
from tb_cliente 
where UPPER( tb_cliente.tx_nome) like UPPER( 'a%')
or UPPER( tb_cliente.tx_nome) like UPPER ( 'o%');

WITH media AS (
    SELECT
        cd_categoria,
        AVG(nr_estoque) AS vl_media_categoria
    FROM tb_produto
    GROUP BY cd_categoria
)
SELECT *
FROM tb_produto
INNER JOIN media
    ON tb_produto.cd_categoria = media.cd_categoria
INNER JOIN tb_categoria
    ON tb_produto.cd_categoria = tb_categoria.id_categoria
WHERE tb_produto.nr_estoque < media.vl_media_categoria;

with vendas as (
select
count(distinct venda.id_venda) as nr_compras,
max(venda.dt_venda ) as dt_ultimacompra,
min(venda.dt_venda ) as dt_primeiracompra,
sum(prod.nr_quantidade * prod.vl_unitario) as vl_totalgasto,
venda.cd_cliente as cliente
from tb_venda venda
inner join tb_item_venda prod
on prod.cd_venda = venda.id_venda
where
upper(venda.tx_status) = upper('pago')
group by venda.cd_cliente
)
select
cliente.tx_nome as nome,
ende.tx_cidade as cidade,
vendas.dt_primeiracompra,
vendas.dt_ultimacompra,
vendas.vl_totalgasto,
vendas.nr_compras,
case
when vendas.nr_compras < 1 then 'CLIENTE PEQUENO'
when vendas.nr_compras = 1 then 'CLIENTE MEDIO'
when vendas.nr_compras = 2 then 'CLIENTE GRANDE'
else 'OUTROS'
end as status,
(
select
p.tx_nome as produtoFavorito
from
tb_venda venda
inner join tb_item_venda prod on
prod.cd_venda = venda.id_venda
inner join tb_produto p on p.id_produto = prod.cd_produto
where
upper(venda.tx_status) = upper('pago')
and cliente.id_cliente = venda.cd_cliente
group by p.tx_nome, p.id_produto
order by sum(prod.nr_quantidade) desc
limit 1
) as produtoFavorito2
from tb_cliente cliente
inner join tb_endereco ende on ende.id_endereco = cliente.cd_endereco
inner join vendas on vendas.cliente = cliente.id_cliente;
